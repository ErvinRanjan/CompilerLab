decl
int swap(int* a,int* b); 
enddecl 
int swap(int* a,int* b){
    decl
    int temp; 
    enddecl 
    begin 
    temp = *a;
    *a = *b;
    *b = temp;
    return 0;
    end
}
int main(){
    decl 
    int a,b,temp;
    enddecl 
    begin 
    a = 1;
    b = 2;
    temp = swap(&a,&b);
    write(a);
    write(b);
    return 0;
    end
}
