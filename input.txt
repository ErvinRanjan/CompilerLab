typedecl 
tuple person(str name,int age);
tuple programmer(person person,int skill);
endtypedecl 
decl 
int personHello(str name,int age);
enddecl 
int personHello(str name,int age){
    decl 
    person p1,p2;
    enddecl
    begin 
    p1.name = name;
    p1.age = age;
    p2 = p1;
    write("hello");
    write("by");
    write(p2.name);
    write(p2.age);
    return 0; 
    end
}
int main(){
    decl 
    person p1,p2;
    programmer p3;
    enddecl
    begin 
    p1.name = "ervin";
    p1.age = 12;
    p3.person = p1;
    p3.skill = 100;
    write("hello");
    write("by");
    write(p3.person.name);
    write(p3.skill);
    return 0; 
    end
}
