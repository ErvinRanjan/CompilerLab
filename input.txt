class Animal{
    decl 
    int makeSound();
    enddecl 
    int makeSound(){
        decl 
        enddecl 
        begin
        write("sound");
        return 0; 
        end 
    }
}
class Dog extends Animal{
    decl 
    int makeSound();
    enddecl 
    int makeSound(){
        decl 
        enddecl 
        begin
        write("bark");
        return 0; 
        end 
    }
}
class Horse extends Animal{
    decl 
    int makeSound();
    enddecl 
    int makeSound(){
        decl 
        enddecl 
        begin
        write("neigh");
        return 0; 
        end 
    }
}
int main(){
    decl 
    Animal* a;
    int temp;
    enddecl 
    begin 
    temp = a->makeSound();
    read(temp);
    if(temp == 0) then 
        a = new(Dog);
    else 
        a = new(Horse);
    endif;
    temp = a->makeSound();
    return 0;
    end
}
