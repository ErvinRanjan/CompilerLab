decl 
int f(int n),sum(int a,int b),swap(int* a,int* b),gcd(int a,int b);
enddecl 
int f(int n){
    decl 
    int ans;
    enddecl 
    begin
    if(n <= 1) then 
        ans = 1;
    else 
        ans = n*f(n - 1);
    endif
    return ans;
    end
}
int sum(int a,int b){
    decl
    int x; 
    enddecl 
    begin
    x = a + b;
    return x;
    end
}
int swap(int* a,int* b){
    decl
    int temp; 
    enddecl
    begin
    temp = *a;
    *a = *b;
    *b = temp;
    return 0;
    end
}
int gcd(int a,int b){
    decl 
    int ans;
    enddecl 
    begin 
    if(b == 0) then 
        ans = a;
    else 
        ans = gcd(b,a%b);
    endif 
    return ans;
    end
}
int main(){
    decl 
    int a,b,temp;
    enddecl
    begin 
    read(a);
    read(b);
    write(gcd(a,b));
    return 0; 
    end
}
