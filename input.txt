decl 
int swap(int* a,int* b),g;
str hello();
int* getRef(int* a);
enddecl 
int swap(int* a,int* b){
    decl
    int temp; 
    enddecl
    begin
    temp = *a;
    *a = *b;
    *b = temp;
    return 0;
    end
}
str hello(){
    decl 
    str h;
    enddecl
    begin 
    h = "hello";
    return h; 
    end
}
int* getRef(int* a){
    decl
    int* ans; 
    enddecl 
    begin 
    ans = a;
    return ans;
    end
}
int main(){
    decl
    int a,b,temp,*p;
    enddecl
    begin
    g = 1; 
    p = getRef(&g); // experimental function to test return types
    *p = 3;
    write(g);
    a = 1;
    b = 2;
    temp = swap(&a,&b);
    write(a);
    write(b);
    return 0; 
    end
}
